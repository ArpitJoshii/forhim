export default { async fetch(request, env, ctx) { const url = new URL(request.url); const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Max-Age": "86400" }; // ------------------------------------------------------ // CORS preflight // ------------------------------------------------------ if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } if (request.method !== "GET") { return json({ ok: false, error: "GET requests only" }, 405, cors); } // ------------------------------------------------------ // Page // ------------------------------------------------------ const page = url.searchParams.get("page"); if (!page) { return json({ ok: false, error: "Missing page parameter" }, 400, cors); } // ------------------------------------------------------ // Security // ------------------------------------------------------ if (!/^[a-zA-Z0-9_-]+$/.test(page)) { return json({ ok: false, error: "Invalid Telegraph page" }, 400, cors); } // ------------------------------------------------------ // Telegraph API // ------------------------------------------------------ const apiURL = "https://api.telegra.ph/getPage/" + encodeURIComponent(page) + "?return_content=true"; try { /* * Cache the Telegraph response. * * This means repeatedly opening the same gallery * doesn't necessarily require another Telegraph * request every time. */ const cacheKey = new Request( "https://gallery-cache/" + page ); const cache = caches.default; let response = await cache.match(cacheKey); if (response) { return new Response( response.body, { status: response.status, headers: { ...Object.fromEntries( response.headers ), ...cors } } ); } // -------------------------------------------------- // Fetch Telegraph // -------------------------------------------------- const telegraphResponse = await fetch(apiURL, { headers: { "Accept": "application/json" } }); if (!telegraphResponse.ok) { return json({ ok: false, error: "Telegraph API error", status: telegraphResponse.status, page }, 502, cors); } const data = await telegraphResponse.json(); if (!data.ok) { return json({ ok: false, error: data.error || "Telegraph page not found", page }, 404, cors); } // -------------------------------------------------- // Cache successful response // -------------------------------------------------- const output = new Response( JSON.stringify(data), { status: 200, headers: { "Content-Type": "application/json; charset=UTF-8", "Cache-Control": "public, max-age=1800" } } ); ctx.waitUntil( cache.put( cacheKey, output.clone() ) ); return new Response( output.body, { status: 200, headers: { "Content-Type": "application/json; charset=UTF-8", "Cache-Control": "public, max-age=1800", ...cors } } ); } catch (error) { return json({ ok: false, error: "Unable to contact Telegraph", details: error.message }, 500, cors); } } }; // ============================================================ // JSON HELPER // ============================================================ function json(data, status, cors) { return new Response( JSON.stringify(data), { status, headers: { "Content-Type": "application/json; charset=UTF-8", ...cors } } ); }
